#!/usr/bin/env bash
# =============================================================================
# FLOPPY-COGNITION BBS  —  Science-grade terminal control plane
# =============================================================================
# A BBS-style operator shell for the pure-Python cognition stack.
# 48 cutting-edge approaches are implemented as first-class operators
# (see menu [48] / docs/BBS-48.md). Stdlib bash + python3 only.
#
# Usage:
#   chmod +x floppy-bbs.sh
#   ./floppy-bbs.sh
#   ./floppy-bbs.sh --once train-status
#   ./floppy-bbs.sh --cmd health
# =============================================================================
set -euo pipefail

ROOT="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
cd "$ROOT"

# ---------- approach 01: deterministic session identity ----------
SESSION_ID="$(date -u +%Y%m%dT%H%M%SZ)-$$"
LOG_DIR="${ROOT}/logs/bbs"
mkdir -p "$LOG_DIR" models logs
SESSION_LOG="${LOG_DIR}/session-${SESSION_ID}.jsonl"
METRICS_DIR="${LOG_DIR}/metrics"
mkdir -p "$METRICS_DIR"

# ---------- approach 02: ANSI semantic palette (BBS phosphor) ----------
if [[ -t 1 ]]; then
  C_RESET=$'\033[0m'; C_BOLD=$'\033[1m'; C_DIM=$'\033[2m'
  C_GREEN=$'\033[38;5;46m'; C_CYAN=$'\033[38;5;51m'; C_YEL=$'\033[38;5;226m'
  C_MAG=$'\033[38;5;201m'; C_RED=$'\033[38;5;196m'; C_BLU=$'\033[38;5;39m'
  C_ORG=$'\033[38;5;208m'; C_WHT=$'\033[38;5;255m'
else
  C_RESET=; C_BOLD=; C_DIM=; C_GREEN=; C_CYAN=; C_YEL=; C_MAG=; C_RED=; C_BLU=; C_ORG=; C_WHT=
fi

# ---------- approach 03: structured event bus (JSONL) ----------
bbs_log() {
  local kind="$1"; shift
  local msg="$*"
  local ts; ts="$(date -u +%Y-%m-%dT%H:%M:%SZ)"
  printf '{"ts":"%s","session":"%s","kind":"%s","msg":%s}\n' \
    "$ts" "$SESSION_ID" "$kind" "$(python3 -c 'import json,sys; print(json.dumps(sys.argv[1]))' "$msg")" \
    >> "$SESSION_LOG" 2>/dev/null || true
}

# ---------- approach 04: adaptive banner / telemetry header ----------
bbs_header() {
  local cpu mem models
  cpu="$(nproc 2>/dev/null || echo '?')"
  mem="$(python3 -c \"import os; print('%.0f'%(os.sysconf('SC_PAGE_SIZE')*os.sysconf('SC_PHYS_PAGES')/1e9 if hasattr(os,'sysconf') else 0))\" 2>/dev/null || echo '?')"
  models="$(find models -name '*.riskytensor' 2>/dev/null | wc -l | tr -d ' ')"
  clear 2>/dev/null || true
  echo -e "${C_CYAN}${C_BOLD}"
  cat <<'BANNER'
  +==================================================================+
  |   F L O P P Y - C O G N I T I O N   B B S   -   ops plane v1    |
  |   pure-python - neuro - fmc - arena - riskytensor - science     |
  +==================================================================+
BANNER
  echo -e "${C_RESET}${C_DIM}  session ${SESSION_ID}  -  cores ${cpu}  -  host_ram_gb ~${mem}  -  models ${models}${C_RESET}"
  echo -e "${C_DIM}  root ${ROOT}${C_RESET}"
  echo
}

# ---------- approach 05: pure-function menu registry ----------
declare -a MENU_IDS MENU_LABELS MENU_FNS
bbs_register() {
  MENU_IDS+=("$1"); MENU_LABELS+=("$2"); MENU_FNS+=("$3")
}

# ---------- approach 06: safe python runner (isolated -I) ----------
py() { python3 -I "$@"; }
py_mod() { export PYTHONPATH="$ROOT${PYTHONPATH:+:$PYTHONPATH}"; python3 -I "$@"; }

# ---------- approach 07: model inventory with science metadata ----------
cmd_models() {
  echo -e "${C_YEL}${C_BOLD}> Model lattice${C_RESET}"
  py_mod - <<'PY'
import os, json, sys
from pathlib import Path
sys.path.insert(0, ".")
root = Path(".")
rows = []
for p in sorted(root.glob("**/*.riskytensor")):
    meta = p.with_suffix(p.suffix + ".meta.json")
    if not meta.exists():
        meta = Path(str(p).replace(".riskytensor", ".meta.json"))
    info = {"path": str(p).replace("\\\\","/"), "bytes": p.stat().st_size}
    if meta.exists():
        try: info.update(json.loads(meta.read_text()))
        except Exception: pass
    rows.append(info)
if not rows:
    print("  (no .riskytensor found under models/)")
else:
    for r in rows:
        mib = r["bytes"] / 1048576
        cfg = r.get("cfg") or {}
        focus = r.get("focus") or r.get("entity") or ""
        print(f"  {r['path']:<42} {mib:6.3f} MiB  params={r.get('params') or r.get('params_measured') or '?'}  {cfg}  {focus}")
PY
  bbs_log models "listed"
}

# ---------- approach 08: health / gate G-0 style probe ----------
cmd_health() {
  echo -e "${C_GREEN}${C_BOLD}> System health (compile + imports + util)${C_RESET}"
  local ok=0
  for m in train2 neuro fmc util_opt riskytensor bumpy laser; do
    if PYTHONPATH="$ROOT" python3 -I -c "import sys; sys.path.insert(0,'$ROOT'); import $m" 2>/dev/null; then
      echo -e "  ${C_GREEN}OK${C_RESET}  import $m"
    else
      echo -e "  ${C_RED}FAIL${C_RESET} import $m"; ok=1
    fi
  done
  py_mod -m py_compile dashboard_v8.py client_v7.py train2.py neuro.py 2>/dev/null \
    && echo -e "  ${C_GREEN}OK${C_RESET}  py_compile core" \
    || { echo -e "  ${C_RED}FAIL${C_RESET} py_compile"; ok=1; }
  if [[ -f util_opt.py ]]; then
    py_mod -c "import util_opt,json; print('  util', json.dumps(util_opt.report()))" 2>/dev/null || true
  fi
  bbs_log health "ok=$ok"
  return $ok
}

# ---------- approach 09: neuro field smoke (deterministic) ----------
cmd_neuro_smoke() {
  echo -e "${C_MAG}${C_BOLD}> Neuro field smoke (collapse vs clean)${C_RESET}"
  py_mod - <<PY
import sys; sys.path.insert(0, r"$ROOT")
import neuro
class St: pass
st=St(); st.last_reward=0; st.temp=0.3; st.topk=5; st.max_new=16
st.neuro_state=None; st.neuro_engine=None; st.last_neuro=None
a=neuro.neuro_for_answer(st, ans="I s is.", secs=1.0, collapse=True, pattern="residual")
b=neuro.neuro_for_answer(st, ans="Hello is a greeting.", secs=1.0, collapse=False)
assert a is not None and b is not None
print("  collapse domains:", list((a.get("domains") or {}).keys())[:6], "...")
print("  collapse_risk:", (a.get("flags") or {}).get("collapse_risk"))
print("  clean domains n:", len(b.get("domains") or {}))
print("  OK — neuro never returns bare None")
PY
  bbs_log neuro_smoke "ok"
}
